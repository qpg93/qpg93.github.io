{{- $key := .Get 0 -}} {{- $chart := index .Page.Params.charts $key -}} {{- if not $chart -}} {{- errorf "Missing chart %q in %s" $key .Page.File.Path -}} {{- else -}} {{- $id := printf "chart-%s-%d" .Page.File.UniqueID .Ordinal -}} {{- $max := float $chart.max -}} {{- $xmax := float $chart.xmax -}} {{- $coordinates := slice -}} {{- range $chart.points -}} {{- $x := add 72 (mul 568 (div (float .x) $xmax)) -}} {{- $y := sub 250 (mul 208 (div (float .value) $max)) -}} {{- $coordinates = $coordinates | append (printf "%.2f,%.2f" $x $y) -}} {{- end -}} {{- end -}}
